from google.colab import files
import pandas as pd
import io

from plotnine import (
    ggplot, aes, geom_segment, geom_point, geom_text,
    labs, theme_minimal, theme, element_text, element_blank,
    scale_color_manual
)

# 1. Upload do arquivo


uploaded = files.upload()

nome_arquivo = list(uploaded.keys())[0]

df = pd.read_csv(
    io.BytesIO(uploaded[nome_arquivo])
)

print("Arquivo carregado:", nome_arquivo)

# 2. Soma dos atributos de cada Pokémon

atributos = [
    "hp",
    "attack",
    "defense",
    "sp atk",
    "sp def",
    "speed"
]

df["total_stats"] = df[atributos].sum(axis=1)


# 3. Resumo por tipo principal

resumo = (
    df.groupby("type")
      .agg(
          media=("total_stats", "mean"),
          quantidade=("name", "count")
      )
      .reset_index()
)

# 4. Tradução dos tipos
traducao = {
    "normal": "Normal",
    "fire": "Fogo",
    "water": "Água",
    "electric": "Elétrico",
    "grass": "Planta",
    "ice": "Gelo",
    "fighting": "Lutador",
    "poison": "Veneno",
    "ground": "Terrestre",
    "flying": "Voador",
    "psychic": "Psíquico",
    "bug": "Inseto",
    "rock": "Pedra",
    "ghost": "Fantasma",
    "dragon": "Dragão",
    "dark": "Sombrio",
    "steel": "Aço",
    "fairy": "Fada"
}

resumo["tipo_pt"] = resumo["type"].map(traducao)

# 5. Adiciona quantidade no nome da categoria

resumo["tipo_label"] = (
    resumo["tipo_pt"]
    + " (n = "
    + resumo["quantidade"].astype(str)
    + ")"
)

# 6. Ordena pela média

resumo = resumo.sort_values("media")

resumo["tipo_label"] = pd.Categorical(
    resumo["tipo_label"],
    categories=resumo["tipo_label"],
    ordered=True
)

# 7. Cores características dos tipos
cores_tipos = {
    "Normal": "#A8A77A",
    "Fogo": "#EE8130",
    "Água": "#6390F0",
    "Elétrico": "#F7D02C",
    "Planta": "#7AC74C",
    "Gelo": "#96D9D6",
    "Lutador": "#C22E28",
    "Veneno": "#A33EA1",
    "Terrestre": "#E2BF65",
    "Voador": "#A98FF3",
    "Psíquico": "#F95587",
    "Inseto": "#A6B91A",
    "Pedra": "#B6A136",
    "Fantasma": "#735797",
    "Dragão": "#6F35FC",
    "Sombrio": "#705746",
    "Aço": "#B7B7CE",
    "Fada": "#D685AD"
}

# Como agora a variável de cor será "tipo_label",
# criamos um dicionário correspondente
cores = {
    linha["tipo_label"]: cores_tipos[linha["tipo_pt"]]
    for _, linha in resumo.iterrows()
}

# 8. Valores auxiliares

resumo["inicio"] = resumo["media"].min() - 25

resumo["valor"] = resumo["media"].map(
    lambda x: f"{x:.1f}"
)


# 9. Construção do gráfico

grafico = (
    ggplot(
        resumo,
        aes(
            x="media",
            y="tipo_label",
            color="tipo_label"
        )
    )

    # Linha horizontal
    + geom_segment(
        aes(
            x="inicio",
            xend="media",
            y="tipo_label",
            yend="tipo_label"
        ),
        color="#D9D9D9",
        size=1.2
    )

    # Pontos com tamanho fixo
    + geom_point(
        size=6,
        alpha=0.95
    )

    # Valor da média
    + geom_text(
        aes(label="valor"),
        nudge_x=7,
        color="#333333",
        size=9
    )

    # Cores
    + scale_color_manual(
        values=cores
    )

    # Títulos
    + labs(
        title="Pontuação média dos Pokémon por Espécie",
        subtitle=(
            "Soma média de HP, Ataque, Defesa, Ataque Especial, "
            "Defesa Especial e Velocidade"
        ),
        x="Pontuação média total dos atributos",
        y="Tipo principal e quantidade de Pokémon",
        caption="Fonte: Pokemon_full.csv"
    )

    # Tema
    + theme_minimal()

    + theme(
        figure_size=(12, 8),

        plot_title=element_text(
            size=18,
            weight="bold"
        ),

        plot_subtitle=element_text(
            size=11,
            color="#555555"
        ),

        axis_title=element_text(
            size=11,
            weight="bold"
        ),

        axis_text_y=element_text(
            size=10,
            weight="bold"
        ),

        axis_text_x=element_text(
            size=9
        ),

        panel_grid_major_y=element_blank(),

        panel_grid_minor=element_blank(),

        # Remove a legenda lateral,
        # pois os tipos já aparecem no eixo Y
        legend_position="none"
    )
)

grafico
