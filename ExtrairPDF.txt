import pdfplumber
import pandas as pd
import re

arquivo_pdf = "cadastro.pdf"

# Extração do texto do PDF
paginas = []

with pdfplumber.open(arquivo_pdf) as pdf:
    for pagina in pdf.pages:
        texto_pagina = pagina.extract_text()

        if texto_pagina:
            paginas.append(texto_pagina)

texto = "\n".join(paginas)


# Separação dos cadastros
blocos = re.split(
    r'(?im)(?=^\s*nome\s*:)',
    texto
)

blocos = [
    bloco.strip()
    for bloco in blocos
    if re.match(r'(?is)^\s*nome\s*:', bloco)
]


# Função para extrair os campos
def extrair_campo(bloco, padrao):
    resultado = re.search(
        padrao,
        bloco,
        flags=re.IGNORECASE | re.MULTILINE
    )

    if resultado:
        return resultado.group(1).strip()

    return None


# Padronização da data
def formatar_data(data):
    if not data:
        return None

    data = data.strip()

    meses = {
        "jan": "01",
        "feb": "02",
        "mar": "03",
        "apr": "04",
        "may": "05",
        "jun": "06",
        "jul": "07",
        "aug": "08",
        "sep": "09",
        "oct": "10",
        "nov": "11",
        "dec": "12"
    }

    for mes_texto, mes_numero in meses.items():
        data = re.sub(
            rf'\b{mes_texto}\b',
            mes_numero,
            data,
            flags=re.IGNORECASE
        )

    data = data.replace("-", "/")
    data = re.sub(r'\s+', '', data)

    return data


# Padronização do CEP
def formatar_cep(cep):
    if not cep:
        return None

    numeros = re.sub(r'\D', '', cep)

    if len(numeros) == 8:
        return f"{numeros[:5]}-{numeros[5:]}"

    return cep.strip()


# Padronização do CPF
def formatar_cpf(cpf):
    if not cpf:
        return None

    numeros = re.sub(r'\D', '', cpf)

    if len(numeros) == 11:
        return f"{numeros[:3]}.{numeros[3:6]}.{numeros[6:9]}-{numeros[9:]}"

    return cpf.strip()


# Padronização do telefone
def formatar_telefone(telefone):
    if not telefone:
        return None

    numeros = re.sub(r'\D', '', telefone)

    if len(numeros) == 11:
        return f"({numeros[:2]}) {numeros[2:7]}-{numeros[7:]}"

    if len(numeros) == 10:
        return f"({numeros[:2]}) {numeros[2:6]}-{numeros[6:]}"

    return telefone.strip()


# Extração e organização dos dados
dados = []

for bloco in blocos:

    nome_completo = extrair_campo(
        bloco,
        r'^\s*nome\s*:\s*(.+)$'
    )

    nome = nome_completo
    apelido = None

    if nome_completo:
        resultado_aka = re.match(
            r'^(.*?)\s*\(\s*aka\s+(.+?)\s*\)\s*$',
            nome_completo,
            flags=re.IGNORECASE
        )

        if resultado_aka:
            nome = resultado_aka.group(1).strip()
            apelido = resultado_aka.group(2).strip()

    data_nascimento = extrair_campo(
        bloco,
        r'^\s*(?:data\s+de\s+nascimento|dt\s*nasc)\s*:\s*(.+)$'
    )

    endereco = extrair_campo(
        bloco,
        r'^\s*endere[cç]o\s*:\s*(.+)$'
    )

    cep = extrair_campo(
        bloco,
        r'^\s*cep\s*:\s*([\d.\-]+)\s*$'
    )

    if not cep and endereco:
        resultado_cep = re.search(
            r'\bCEP\s*:\s*([\d.\-]+)',
            endereco,
            flags=re.IGNORECASE
        )

        if resultado_cep:
            cep = resultado_cep.group(1)

    if endereco:
        endereco = re.sub(
            r'\s*CEP\s*:\s*[\d.\-]+\s*$',
            '',
            endereco,
            flags=re.IGNORECASE
        ).strip()

    telefone = extrair_campo(
        bloco,
        r'^\s*(?:tel|telefone)\s*:\s*(.+)$'
    )

    cpf = extrair_campo(
        bloco,
        r'^\s*cpf\s*:\s*(.+)$'
    )

    dados.append({
        "Nome": nome,
        "Apelido": apelido,
        "Data de nascimento": formatar_data(data_nascimento),
        "CEP": formatar_cep(cep),
        "Endereço": endereco,
        "Telefone": formatar_telefone(telefone),
        "CPF": formatar_cpf(cpf)
    })


# Criação do DataFrame
df = pd.DataFrame(dados)

print(df)
